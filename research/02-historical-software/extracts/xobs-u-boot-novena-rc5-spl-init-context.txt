   527	#ifdef CONFIG_BOARD_POSTCLK_INIT
   528		board_postclk_init();
   529	#endif
   530	#ifdef CONFIG_FSL_ESDHC
   531		get_clocks();
   532	#endif

   534		/* Setup IOMUX and configure basics. */
   535		novena_spl_setup_iomux_audio();
   536		novena_spl_setup_iomux_buttons();
   537		novena_spl_setup_iomux_enet();
   538		novena_spl_setup_iomux_fpga();
   539		novena_spl_setup_iomux_i2c();
   540		novena_spl_setup_iomux_pcie();
   541		novena_spl_setup_iomux_sdhc();
   542		novena_spl_setup_iomux_spi();
   543		novena_spl_setup_iomux_uart();
   544		novena_spl_setup_iomux_video();

   546		/* UART clocks enabled and gd valid - init serial console */
   547		preloader_console_init();
   548	
   549		/* Start the DDR DRAM */
   550	
   551		novena_read_spd(&novena_ddr_info, &novena_ddr3_cfg);
   552		mx6dq_dram_iocfg(novena_ddr3_cfg.width,

   556		do_dqs_calibration();

   559			printf("Fail!\n");
