   270	/* setup board specific PMIC */
   271	int power_init_board(void)
   272	{
   273		struct pmic *p;
   274		u32 reg;
   275		int ret;
   276	
   277		power_pfuze100_init(1);
   278		p = pmic_get("PFUZE100");
   279		if (!p)
   280			return -EINVAL;
   281	
   282		ret = pmic_probe(p);
   283		if (ret)
   284			return ret;
   285	
   286		pmic_reg_read(p, PFUZE100_DEVICEID, &reg);
   287		printf("PMIC:  PFUZE100 ID=0x%02x\n", reg);
   288	
   289		/* Set SWBST to 5.0V and enable (for USB) */
   290		pmic_reg_read(p, PFUZE100_SWBSTCON1, &reg);
   291		reg &= ~(SWBST_MODE_MASK | SWBST_VOL_MASK);
   292		reg |= (SWBST_5_00V | SWBST_MODE_AUTO);
   293		pmic_reg_write(p, PFUZE100_SWBSTCON1, reg);
   294	
   295		return 0;
   296	}

   298	/* EEPROM configuration data */
   299	struct novena_eeprom_data {
   300		uint8_t		signature[6];
   301		uint8_t		version;
   302		uint8_t		reserved;
   303		uint32_t	serial;
   304		uint8_t		mac[6];
   305		uint16_t	features;
   306	};
   307	
   308	int misc_init_r(void)
   309	{
   310		struct novena_eeprom_data data;
   311		uchar *datap = (uchar *)&data;
   312		const char *signature = "Novena";
   313		int ret;
   314	
   315		/* If 'ethaddr' is already set, do nothing. */
   316		if (getenv("ethaddr"))
   317			return 0;
   318	
   319		/* EEPROM is at bus 2. */
   320		ret = i2c_set_bus_num(2);
   321		if (ret) {
   322			puts("Cannot select EEPROM I2C bus.\n");
   323			return 0;
   324		}
   325	
   326		/* EEPROM is at address 0x56. */
   327		ret = eeprom_read(0x56, 0, datap, sizeof(data));
   328		if (ret) {
   329			puts("Cannot read I2C EEPROM.\n");
   330			return 0;
   331		}
   332	
   333		/* Check EEPROM signature. */
   334		if (memcmp(data.signature, signature, 6)) {
   335			puts("Invalid I2C EEPROM signature.\n");
   336			return 0;
   337		}
   338	
   339		/* Set ethernet address from EEPROM. */
   340		eth_setenv_enetaddr("ethaddr", data.mac);
   341	
   342		return ret;
   343	}
