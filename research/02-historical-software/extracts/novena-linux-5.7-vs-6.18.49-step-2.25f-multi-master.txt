static int i2c_imx_bus_busy(struct imx_i2c_struct *i2c_imx, int for_busy, bool atomic)
{
	unsigned long orig_jiffies = jiffies;
	unsigned int temp;

	dev_dbg(&i2c_imx->adapter.dev, "<%s>\n", __func__);

	while (1) {
		temp = imx_i2c_read_reg(i2c_imx, IMX_I2C_I2SR);

		/* check for arbitration lost */
		if (temp & I2SR_IAL) {
			temp &= ~I2SR_IAL;
			imx_i2c_write_reg(temp, i2c_imx, IMX_I2C_I2SR);
			return -EAGAIN;
		}

		if (for_busy && (temp & I2SR_IBB)) {
			i2c_imx->stopped = 0;
			break;
		}
		if (!for_busy && !(temp & I2SR_IBB)) {
			i2c_imx->stopped = 1;
			break;
		}
		if (time_after(jiffies, orig_jiffies + msecs_to_jiffies(500))) {
			dev_dbg(&i2c_imx->adapter.dev,
				"<%s> I2C bus is busy\n", __func__);
			return -ETIMEDOUT;
		}
		if (atomic)
			udelay(100);
		else
			schedule();
	}

	return 0;
}

static int i2c_imx_bus_busy(struct imx_i2c_struct *i2c_imx, int for_busy, bool atomic)
{
	bool multi_master = i2c_imx->multi_master;
	unsigned long orig_jiffies = jiffies;
	unsigned int temp;

	while (1) {
		temp = imx_i2c_read_reg(i2c_imx, IMX_I2C_I2SR);

		/* check for arbitration lost */
		if (multi_master && (temp & I2SR_IAL)) {
			dev_err(&i2c_imx->adapter.dev,
				"NOVENA-I2C: arbitration lost in bus_busy, I2SR=0x%02x\n",
				temp);

			i2c_imx_clear_irq(i2c_imx, I2SR_IAL);
			return -EAGAIN;
		}

		if (for_busy && (!multi_master || (temp & I2SR_IBB))) {
			i2c_imx->stopped = 0;
			break;
		}
		if (!for_busy && !(temp & I2SR_IBB)) {
			i2c_imx->stopped = 1;
			break;
		}
		if (time_after(jiffies, orig_jiffies + msecs_to_jiffies(500))) {
			dev_dbg(&i2c_imx->adapter.dev,
				"<%s> I2C bus is busy\n", __func__);
			return -ETIMEDOUT;
		}
		if (atomic)
			udelay(100);
		else
			schedule();
	}

	return 0;
}

static int i2c_imx_start(struct imx_i2c_struct *i2c_imx, bool atomic)
{
	unsigned int temp = 0;
	int result;

	dev_dbg(&i2c_imx->adapter.dev, "<%s>\n", __func__);

	imx_i2c_write_reg(i2c_imx->ifdr, i2c_imx, IMX_I2C_IFDR);
	/* Enable I2C controller */
	imx_i2c_write_reg(i2c_imx->hwdata->i2sr_clr_opcode, i2c_imx, IMX_I2C_I2SR);
	imx_i2c_write_reg(i2c_imx->hwdata->i2cr_ien_opcode, i2c_imx, IMX_I2C_I2CR);

	/* Wait controller to be stable */
	if (atomic)
		udelay(50);
	else
		usleep_range(50, 150);

	/* Start I2C transaction */
	temp = imx_i2c_read_reg(i2c_imx, IMX_I2C_I2CR);
	temp |= I2CR_MSTA;
	imx_i2c_write_reg(temp, i2c_imx, IMX_I2C_I2CR);
	result = i2c_imx_bus_busy(i2c_imx, 1, atomic);
	if (result)
		return result;

	temp |= I2CR_IIEN | I2CR_MTX | I2CR_TXAK;
	if (atomic)
		temp &= ~I2CR_IIEN; /* Disable interrupt */

	temp &= ~I2CR_DMAEN;
	imx_i2c_write_reg(temp, i2c_imx, IMX_I2C_I2CR);
	return result;
}

static int i2c_imx_start(struct imx_i2c_struct *i2c_imx, bool atomic,
			 bool trace_it6251)
{
	unsigned int temp = 0;
	unsigned int i;
	int result;

	imx_i2c_write_reg(i2c_imx->ifdr, i2c_imx, IMX_I2C_IFDR);
	/* Enable I2C controller */
	imx_i2c_write_reg(i2c_imx->hwdata->i2sr_clr_opcode, i2c_imx, IMX_I2C_I2SR);
	imx_i2c_write_reg(i2c_imx->hwdata->i2cr_ien_opcode, i2c_imx, IMX_I2C_I2CR);

	/* Wait controller to be stable */
	if (atomic)
		udelay(50);
	else
		usleep_range(50, 150);

	i2c_imx->novena_trace_it6251 = trace_it6251;

	if (trace_it6251) {
		/* A: controller stable, immediately before requesting MSTA. */
		i2c_imx->novena_i2sr_a =
			imx_i2c_read_reg(i2c_imx, IMX_I2C_I2SR);
		i2c_imx->novena_i2cr_a =
			imx_i2c_read_reg(i2c_imx, IMX_I2C_I2CR);
	}

	/* Start I2C transaction */
	temp = imx_i2c_read_reg(i2c_imx, IMX_I2C_I2CR);
	temp |= I2CR_MSTA;
	imx_i2c_write_reg(temp, i2c_imx, IMX_I2C_I2CR);

	if (trace_it6251) {
		/*
		 * Resolve the MSTA/START transition with a tight sequence of
		 * ordered register reads. There are deliberately no delays,
		 * logging calls or timestamp reads inside this window.
		 */
		for (i = 0; i < NOVENA_IT6251_START_SAMPLES; i++) {
			i2c_imx->novena_start_i2sr[i] =
				imx_i2c_read_reg(i2c_imx, IMX_I2C_I2SR);
			i2c_imx->novena_start_i2cr[i] =
				imx_i2c_read_reg(i2c_imx, IMX_I2C_I2CR);
		}

		/*
		 * Preserve the existing B snapshot: M0 is the first
		 * observation immediately after requesting master mode.
		 */
		i2c_imx->novena_i2sr_b = i2c_imx->novena_start_i2sr[0];
		i2c_imx->novena_i2cr_b = i2c_imx->novena_start_i2cr[0];
	}

	result = i2c_imx_bus_busy(i2c_imx, 1, atomic);

	if (trace_it6251) {
		/* C: immediately after the bus-busy check. */
		i2c_imx->novena_i2sr_c =
			imx_i2c_read_reg(i2c_imx, IMX_I2C_I2SR);
		i2c_imx->novena_i2cr_c =
			imx_i2c_read_reg(i2c_imx, IMX_I2C_I2CR);
	}

	if (result) {
		if (trace_it6251)
			dev_err(&i2c_imx->adapter.dev,
				"NOVENA-I2C: IT6251 START failure "
				"A=%02x/%02x B=%02x/%02x C=%02x/%02x ret=%d\n",
				i2c_imx->novena_i2sr_a, i2c_imx->novena_i2cr_a,
				i2c_imx->novena_i2sr_b, i2c_imx->novena_i2cr_b,
				i2c_imx->novena_i2sr_c, i2c_imx->novena_i2cr_c,
				result);

		if (trace_it6251)
			i2c_imx_novena_dump_start_transition(i2c_imx);

		return result;
	}

	temp |= I2CR_IIEN | I2CR_MTX | I2CR_TXAK;
	if (atomic)
		temp &= ~I2CR_IIEN; /* Disable interrupt */

	temp &= ~I2CR_DMAEN;
	imx_i2c_write_reg(temp, i2c_imx, IMX_I2C_I2CR);

	if (trace_it6251) {
		/* D: controller configured for the first address transfer. */
		i2c_imx->novena_i2sr_d =
			imx_i2c_read_reg(i2c_imx, IMX_I2C_I2SR);
		i2c_imx->novena_i2cr_d =
			imx_i2c_read_reg(i2c_imx, IMX_I2C_I2CR);
	}

	return result;
}
